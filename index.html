function doPost(e) { try { const data = JSON.parse(e.postData.contents); // 기본 수신자 이메일 (CGV, HS마켓 외 업체용) const DEFAULT_RECIPIENT = "기본수신자이메일@example.com"; // 업체별 지정 수신자가 오면 최우선 적용 const recipient = (data.recipientEmail || data.email || data.to || "").trim() || DEFAULT_RECIPIENT; const subject = `[일일안전점검] ${data.company} (${data.boothCount || 1}개 가판) - ${data.inspector}`; const body = `점검 업체: ${data.company}\n` + `점검 가판: ${data.booth}\n` + `점검자: ${data.inspector}\n` + `점검 일시: ${data.date}\n` + `수신 이메일: ${recipient}`; const options = { to: recipient, subject: subject, body: body }; if (data.pdfBase64 && data.fileName) { const bytes = Utilities.base64Decode(data.pdfBase64); options.attachments = [ Utilities.newBlob(bytes, data.mimeType || "application/pdf", data.fileName) ]; } MailApp.sendEmail(options); return ContentService .createTextOutput(JSON.stringify({ ok: true, recipient: recipient })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ ok: false, error: err.message })) .setMimeType(ContentService.MimeType.JSON); } }
